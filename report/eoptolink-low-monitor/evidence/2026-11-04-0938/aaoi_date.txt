							filer: filer,
							chart: false,

						this.update_data(index, ticker);

					if (filer && webDriver.jQuery("#ir_featurebox_"+index+"-filings").length)
						this.update_filings(index, filer);

						this.update_labels(index, ticker, filer);
				},

					var item_id = ticker || filer;

<div class="wd_body wd_news_body fr-view" wd_resize="formatNews" wd_print_url="https://investors.ao-inc.com/2026-10-15-Applied-Optoelectronics-Announces-Date-of-Third-Quarter-2026-Financial-Results-Conference-Call?printable=1"><p><span class="legendSpanClass">SUGAR LAND, Texas</span>, <span class="legendSpanClass">Oct. 15, 2026</span> /<a href="http://www.prnewswire.com/" target="_blank">PRNewswire</a>/ -- <b>Applied Optoelectronics, Inc. </b>(NASDAQ: AAOI)<b> (&quot;AOI&quot;),</b>&nbsp;a leading provider of advanced optical and HFC networking products that power AI, today announced that it will release financial results for its third quarter ended September 30, 2026, on Thursday, November 5, 2026.&nbsp;</p>

<div class="PRN_ImbeddedAssetReference" id="DivAssetPlaceHolder1">
<p><img alt="Applied Optoelectronics, Inc." src="https://mmx.prnewswire.com/media/MS1995467/aoi-corp-full-color-rgb-900px-w-72ppi.jpg?id=OA3022438" title="Applied Optoelectronics, Inc." /></p>
</div>

<p>AOI will host a conference call and webcast for analysts and investors to discuss its third quarter 2026 financial results and outlook for its fourth quarter 2026 at 4:30 p.m. Eastern Time / 3:30 p.m. Central Time the same day. To participate in the call, please dial 888-880-3330 and ask to be joined to the AOI call.</p>

<p>A live audio webcast of the conference call and supplemental financials will be accessible from the company&#39;s Investor Relations website at <a aria-invalid="true" href="http://investors.ao-inc.com/" rel="nofollow" target="_blank">investors.ao-inc.com</a>. Following the webcast, an archived version will be available on the website for one year. A telephonic replay of the call will be available one hour after the call and will be available until November 12, 2026, at 11:59 p.m. Eastern Time / 10:59 p.m. Central Time and may be accessed by dialing (800) 770-2030&nbsp;and entering passcode 4754976 #.</p>

<p><b>About AOI</b> &nbsp;<br />
Applied Optoelectronics, Inc. (AOI) is a leading developer and manufacturer of advanced optical and HFC networking products that serve as the critical building blocks for AI datacenters, CATV and broadband fiber access networks worldwide. AOI supplies these critical infrastructure products to tier-one customers across cloud computing, CATV broadband, telecommunications, and FTTH markets. The company has R&amp;D facilities in Atlanta, GA, and engineering and manufacturing facilities at its corporate headquarters in Sugar Land, TX, as well as in Taipei, Taiwan and Ningbo, China.&nbsp;For additional information, visit&nbsp;<a aria-invalid="true" href="https://www.globenewswire.com/Tracker?data=Zuv2cCxYhyEEudGiXn_XZtJRHkyFClVVyI2vCOEs4nVawRnLI3165angexqJXioyoZUiLBEulGOp5ZAyJNCAJg==" rel="nofollow" target="_blank">www.ao-inc.com</a>.&nbsp;&nbsp;</p>
